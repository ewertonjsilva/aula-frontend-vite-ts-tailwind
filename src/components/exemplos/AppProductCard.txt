const PRODUCTS_DATA = [

    price: 89.90,
    category: "Periféricos",
    isAvailable: true,
    tags: ["gaming", "setup", "hot-swappable"]
  },
  {
    id: 2,
    title: "Rato Óptico Ergonómico 16000 DPI",
    price: 45.00,
    category: "Periféricos",
    isAvailable: false
    // O 'tags' foi omitido aqui para testar a prop opcional
  },
  {
    id: 3,
    title: "Monitor Gaming 27\" IPS 144Hz",
    price: 249.99,
    category: "Monitores",
    isAvailable: true,
    tags: ["4k", "gaming"]
  }
];

function App() {
  // Esta função será executada quando o botão do componente filho for clicado
  const handleAddToCart = (productId: number, event: MouseEvent<HTMLButtonElement>) => {
    console.log("Evento nativo capturado no App:", event.type);
    alert(`Produto #${productId} adicionado ao carrinho com sucesso!`);
  };

  return (
    <div className="min-h-screen w-screen bg-zinc-950 text-zinc-50 p-8">
      <header className="max-w-6xl mx-auto mb-8">
        <h1 className="text-3xl font-extrabold tracking-tight text-zinc-100">
          Loja de Tecnologia
        </h1>
        <p className="text-zinc-400 text-sm">
          Exemplo prático de componentes reutilizáveis com Props e TypeScript.
        </p>
      </header>

      {/* Grid de Produtos */}
      <main className="max-w-6xl mx-auto grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">
        {PRODUCTS_DATA.map((product) => (
          <ProductCard
            key={product.id}
            id={product.id}
            title={product.title}
            price={product.price}
            category={product.category}
            isAvailable={product.isAvailable}
            tags={product.tags}
            onAddToCart={handleAddToCart} // Passando a função como propriedade
          />
        ))}
      </main>
    </div>
  );
}

export default App;
